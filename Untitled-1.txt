fun firestoreNumber(value: Any?): Double {
    return when (value) {
        is Number -> value.toDouble()
        is String -> value.replace(",", "").trim().toDoubleOrNull() ?: 0.0
        else -> 0.0
    }
}

private fun todayFoodDocumentId(): String {
    return SimpleDateFormat("yyyy-MM-dd", Locale.getDefault()).format(Date())
}

private fun foodDocumentId(date: LocalDate): String =
    if (currentPgId() == DEFAULT_LEGACY_PG_ID) date.toString()
    else "${currentPgId()}_${date}"

private fun foodDisplayDate(date: LocalDate): String {
    return date.format(DateTimeFormatter.ofPattern("dd MMM yyyy", Locale.getDefault()))
}

/**
 * Current authenticated user's property context.
 * pgId is NEVER used as a password or as proof of access.
 * It is loaded from the authenticated user's Firestore profile.
 */
object PgSession {
    var pgId: String = ""
    var role: String = ""

    fun clear() {
        pgId = ""
        role = ""
    }
}

private const val DEFAULT_LEGACY_PG_ID = "pg001"

private fun currentPgId(): String =
    PgSession.pgId.ifBlank { DEFAULT_LEGACY_PG_ID }

private fun pgQuery(
    db: FirebaseFirestore,
    collection: String
): Query {
    return db.collection(collection).whereEqualTo("pgId", currentPgId())
}

private fun withPgId(data: Map<String, Any?>): Map<String, Any?> {
    val result = HashMap<String, Any?>(data)
    result["pgId"] = currentPgId()
    return result
}

private fun setPgSessionFromDocument(document: com.google.firebase.firestore.DocumentSnapshot, role: String): Boolean {
    val pgId = document.getString("pgId")
    if (pgId.isNullOrBlank()) return false
    PgSession.pgId = pgId
    PgSession.role = role
    return true
}

private fun migrateLegacyCollectionToPg001(
    db: FirebaseFirestore,
    collectionName: String,
    onComplete: (Boolean) -> Unit
) {
    db.collection(collectionName).get()
        .addOnSuccessListener { snapshot ->
            val legacyDocs = snapshot.documents.filter { it.getString("pgId").isNullOrBlank() }
            if (legacyDocs.isEmpty()) {
                onComplete(true)
                return@addOnSuccessListener
            }

            val chunks = legacyDocs.chunked(400)

            fun commitChunk(index: Int) {
                if (index >= chunks.size) {
                    onComplete(true)
                    return
                }

                val batch = db.batch()
                chunks[index].forEach { document ->
                    batch.update(
                        document.reference,
                        "pgId",
                        DEFAULT_LEGACY_PG_ID
                    )
                }

                batch.commit()
                    .addOnSuccessListener { commitChunk(index + 1) }
                    .addOnFailureListener { onComplete(false) }
            }

            commitChunk(0)
        }
        .addOnFailureListener { onComplete(false) }
}

private fun migrateLegacyDataToPg001(
    db: FirebaseFirestore,
    onComplete: (Boolean) -> Unit
) {
    val collections = listOf(
        "residents",
        "rooms",
        "rentRecords",
        "foodMenus",
        "complaints",
        "serviceRequests",
        "notices",
        "notifications"
    )

    fun migrateAt(index: Int) {
        if (index >= collections.size) {
            onComplete(true)
            return
        }
        migrateLegacyCollectionToPg001(db, collections[index]) { success ->
            if (!success) {
                onComplete(false)
            } else {
                migrateAt(index + 1)
            }
        }
    }

    migrateAt(0)
}

private fun saveFcmToken() {
    FirebaseMessaging.getInstance()
        .token
        .addOnSuccessListener { token ->
            val uid = FirebaseAuth.getInstance().currentUser?.uid
            if (uid != null && PgSession.role == "resident") {
                FirebaseFirestore.getInstance()
                    .collection("residents")
                    .document(uid)
                    .set(
                        withPgId(mapOf("fcmToken" to token)),
                        SetOptions.merge()
                    )
            }
        }
}

class MainActivity : ComponentActivity() {

    private val notificationPermissionLauncher =
        registerForActivityResult(
            ActivityResultContracts.RequestPermission()
        ) { }

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)

        requestNotificationPermission()
        saveFcmToken()

        setContent {
            ResidentHubApp()
        }
    }

    private fun requestNotificationPermission() {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU) {
            if (
                ContextCompat.checkSelfPermission(
                    this,
                    Manifest.permission.POST_NOTIFICATIONS
                ) != PackageManager.PERMISSION_GRANTED
            ) {
                notificationPermissionLauncher.launch(
                    Manifest.permission.POST_NOTIFICATIONS
                )
            }
        }
    }


}

@Composable
fun ResidentHubApp() {

    var currentScreen by remember { mutableStateOf("loading") }
    var pendingMessage by remember { mutableStateOf("Your account is waiting for approval.") }

    // Keep residents/admins signed in across app restarts. FirebaseAuth persists the
    // authenticated session; we only verify the Firestore role/profile before routing.
    DisposableEffect(Unit) {
        val auth = FirebaseAuth.getInstance()
        val db = FirebaseFirestore.getInstance()
        val user = auth.currentUser

        if (user == null) {
            PgSession.clear()
            currentScreen = "login"
            onDispose { }
        } else {
            db.collection("admins").document(user.uid).get()
                .addOnSuccessListener { adminDoc ->
                    if (adminDoc.exists() && adminDoc.getString("role") == "admin") {
                        if (setPgSessionFromDocument(adminDoc, "admin")) {
                            currentScreen = "adminDashboard"
                        } else {
                            auth.signOut()
                            PgSession.clear()
                            currentScreen = "login"
                        }
                    } else {
                        db.collection("residents").document(user.uid).get()
                            .addOnSuccessListener { residentDoc ->
                                val status = residentDoc.getString("status") ?: "Pending"
                                val approved = residentDoc.getBoolean("approved") ?: false
                                val pgId = residentDoc.getString("pgId").orEmpty()
                                val room = residentDoc.getString("room").orEmpty()
                                val bed = residentDoc.getString("bed").orEmpty()
                                if (approved && status == "Active" && pgId.isNotBlank() && room.isNotBlank() && room != "Not Assigned" && bed.isNotBlank() && bed != "Not Assigned") {
                                    PgSession.pgId = pgId
                                    PgSession.role = "resident"
                                    saveFcmToken()
                                    currentScreen = "dashboard"
                                } else {
                                    pendingMessage = if (status == "Inactive") {
                                        "Your account is inactive. Please contact your PG admin."
                                    } else {
                                        "Your account is waiting for admin approval and PG/room/bed assignment."
                                    }
                                    PgSession.clear()
                                    currentScreen = "pending"
                                }
                            }
                            .addOnFailureListener {
                                auth.signOut()
                                PgSession.clear()
                                currentScreen = "login"
                            }
                    }
                }
                .addOnFailureListener {
                    auth.signOut()
                    PgSession.clear()
                    currentScreen = "login"
                }
            onDispose { }
        }
    }

    var breakfast by remember { mutableStateOf("Poha • Tea • Banana") }

                    Button(
                        onClick = {

                            ) {

                            }

                            if (!emailOrPhone.contains("@")) {
                                message =

                            }

                            val auth = FirebaseAuth.getInstance()
                            val db = FirebaseFirestore.getInstance()
                            message = if (isAdmin)
                                "Signing in as admin..."
                            else
                                "Checking resident account..."

                            auth.signInWithEmailAndPassword(
                                emailOrPhone.trim(),
                                password
                            ).addOnSuccessListener {
                                val uid = auth.currentUser?.uid

                                if (uid == null) {
                                    auth.signOut()
                                    PgSession.clear()
                                    message = "Login failed. Please try again."
                                    return@addOnSuccessListener
                                }

                                if (isAdmin) {
                                    db.collection("admins")
                                        .document(uid)
                                        .get()
                                        .addOnSuccessListener { document ->

                                                document.exists() &&
                                                document.getString("role") == "admin"

            // FOOTER

@Composable
fun CreateAccountScreen(
    onAccountCreated: () -> Unit,
    onBack: () -> Unit
) {
    var fullName by remember { mutableStateOf("") }
    var email by remember { mutableStateOf("") }
    var phone by remember { mutableStateOf("") }
    var password by remember { mutableStateOf("") }
    var joinCode by remember { mutableStateOf("") }
    var message by remember { mutableStateOf("") }

    Column(modifier = Modifier.fillMaxSize().padding(24.dp)) {
        TextButton(onClick = onBack) { Text("← Back") }
        Text("Create Account", fontSize = 28.sp, fontWeight = FontWeight.Bold)
        Spacer(modifier = Modifier.height(8.dp))
        Text("Enter the PG Join Code given to you by the PG admin. Your account will remain pending until approval.", fontSize = 14.sp)
        Spacer(modifier = Modifier.height(20.dp))

        OutlinedTextField(value = fullName, onValueChange = { fullName = it }, label = { Text("Full Name") }, modifier = Modifier.fillMaxWidth())
        Spacer(modifier = Modifier.height(12.dp))
        OutlinedTextField(value = email, onValueChange = { email = it }, label = { Text("Email") }, modifier = Modifier.fillMaxWidth())
        Spacer(modifier = Modifier.height(12.dp))
        OutlinedTextField(value = phone, onValueChange = { phone = it }, label = { Text("Phone Number") }, modifier = Modifier.fillMaxWidth())
        Spacer(modifier = Modifier.height(12.dp))
        OutlinedTextField(value = joinCode, onValueChange = { joinCode = it.uppercase(Locale.getDefault()) }, label = { Text("PG Join Code") }, modifier = Modifier.fillMaxWidth())
        Spacer(modifier = Modifier.height(12.dp))
        OutlinedTextField(value = password, onValueChange = { password = it }, label = { Text("Create Password") }, visualTransformation = PasswordVisualTransformation(), modifier = Modifier.fillMaxWidth())
        Spacer(modifier = Modifier.height(20.dp))

        Button(onClick = {
            if (fullName.isBlank() || email.isBlank() || phone.isBlank() || password.isBlank() || joinCode.isBlank()) {
                message = "Please fill all fields, including the PG Join Code."
            } else if (!email.contains("@")) {
                message = "Please enter a valid email address."
            } else if (password.length < 6) {
                message = "Password must be at least 6 characters."
            } else {
                message = "Checking PG Join Code..."
                val auth = FirebaseAuth.getInstance()
                val db = FirebaseFirestore.getInstance()
                val normalizedJoinCode = joinCode.trim().uppercase(Locale.getDefault())
                db.collection("pgJoinCodes").document(normalizedJoinCode).get()
                    .addOnSuccessListener { codeDoc ->
                        val requestedPgId = codeDoc.getString("pgId").orEmpty()
                        val active = codeDoc.getBoolean("active") ?: false
                        if (!codeDoc.exists() || requestedPgId.isBlank() || !active) {
                            message = "Invalid or inactive PG Join Code."
                            return@addOnSuccessListener
                        }
                        message = "Creating account for the selected PG..."
                        auth.createUserWithEmailAndPassword(email.trim(), password)
                            .addOnSuccessListener { result ->
                                val uid = result.user?.uid
                                if (uid == null) { message = "Account was created, but user ID was unavailable."; return@addOnSuccessListener }
                                val profile = mapOf(
                                    "fullName" to fullName.trim(),
                                    "email" to email.trim(),
                                    "phone" to phone.trim(),
                                    "pgId" to "",
                                    "requestedPgId" to requestedPgId,
                                    "joinCode" to normalizedJoinCode,
                                    "status" to "Pending",
                                    "approved" to false,
                                    "room" to "Not Assigned",
                                    "bed" to "Not Assigned",
                                    "roomType" to "Shared",
                                    "floor" to "-",
                                    "monthlyRent" to 0.0,
                                    "rentDueDay" to 0,
                                    "createdAt" to FieldValue.serverTimestamp(),
                                    "updatedAt" to FieldValue.serverTimestamp()
                                )
                                db.collection("residents").document(uid).set(profile)
                                    .addOnSuccessListener {
                                        PgSession.clear()
                                        auth.signOut()
                                        message = "Account created. Your PG admin must approve your account before login."
                                        onAccountCreated()
                                    }
                                    .addOnFailureListener { error -> message = "Profile save failed: ${error.message ?: "Unknown error"}" }
                            }
                            .addOnFailureListener { error -> message = "Account creation failed: ${error.message ?: "Please try again."}" }
                    }
                    .addOnFailureListener { error -> message = "Could not verify PG Join Code: ${error.message ?: "Please try again."}" }
            }
        }, modifier = Modifier.fillMaxWidth()) { Text("Create Account") }

        if (message.isNotEmpty()) {
            Spacer(modifier = Modifier.height(12.dp))
            Text(message, fontSize = 14.sp)
        }
    }
}

@Composable

    var assignedBed by remember { mutableStateOf("Not Assigned") }

    var breakfast by remember { mutableStateOf("Today's breakfast menu") }
    var lunch by remember { mutableStateOf("Today's lunch menu") }
    var dinner by remember { mutableStateOf("Today's dinner menu") }

    DisposableEffect(uid) {
        if (uid == null) {
            onDispose { }
        } else {
            val residentListener = db.collection("residents").document(uid)
                .addSnapshotListener { snapshot, _ ->
                    assignedRoom = snapshot?.getString("room") ?: "Not Assigned"
                    assignedBed = snapshot?.getString("bed") ?: "Not Assigned"
                    residentName = snapshot?.getString("fullName")?.ifBlank { "Resident" } ?: "Resident"
                    residentEmail = snapshot?.getString("email") ?: FirebaseAuth.getInstance().currentUser?.email.orEmpty()
                    val pgId = snapshot?.getString("pgId").orEmpty()
                    if (pgId.isNotBlank()) {
                        db.collection("pgs").document(pgId).get()
                            .addOnSuccessListener { pg ->
                                pgName = pg.getString("name") ?: pg.getString("pgName") ?: pg.getString("displayName") ?: pgId
                            }
                    }
                }
            onDispose { residentListener.remove() }
        }
    }

    DisposableEffect(PgSession.pgId) {
        val pgId = PgSession.pgId
        if (pgId.isBlank()) onDispose { }
        else {
            val foodListener = db.collection("foodMenus").document(foodDocumentId(LocalDate.now()))
                .addSnapshotListener { snap, _ ->
                    if (snap != null && snap.exists()) {
                        breakfast = snap.getString("breakfast")?.ifBlank { "No breakfast menu posted" } ?: "No breakfast menu posted"
                        lunch = snap.getString("lunch")?.ifBlank { "No lunch menu posted" } ?: "No lunch menu posted"

            animation = tween(durationMillis = 12000),

    val navy = Color(0xFF0B1F36)

    val cardSurfaceAlt = Color(0xFF14334B)

    val primaryText = Color(0xFFF3F8FC)
    val secondaryText = Color(0xFFB7C8D8)

    val residentServices = listOf(
        Triple("Rent & Payments", "💳", onRentPayments),
        Triple("Food & Services", "🍽️", onFoodServices),
        Triple("Complaints", "🛠️", onComplaints),
        Triple("Service Requests", "🧰", onFoodServices),
        Triple("Notices", "📢", onNotices),
        Triple("Notifications", "🔔", onNotifications)
    )

    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(
                Brush.verticalGradient(
                    colors = listOf(midnight, navy, Color(0xFF0A2638))
                )
            )
    ) {
        // A very soft teal glow drifts slowly behind the cards; it does not block taps.
        Canvas(modifier = Modifier.fillMaxSize()) {
            val glowX = size.width * (0.18f + 0.64f * glowProgress)
            val glowY = size.height * (0.16f + 0.26f * (1f - glowProgress))

                brush = Brush.radialGradient(
                    colors = listOf(Color(0x3335D0C1), Color.Transparent),

                    radius = size.minDimension * 0.62f

        }
        LazyColumn(
            modifier = Modifier.fillMaxSize().alpha(dashboardAlpha).padding(horizontal = 16.dp, vertical = 12.dp),
            verticalArrangement = Arrangement.spacedBy(14.dp)
        ) {
            item {
                Row(
                    modifier = Modifier.fillMaxWidth(),
                    horizontalArrangement = Arrangement.SpaceBetween,
                    verticalAlignment = Alignment.CenterVertically
                ) {
                    Column {

                    }

@Composable
private fun LaunchedEffectForRentReminder() {
    val uid = FirebaseAuth.getInstance().currentUser?.uid
    DisposableEffect(uid) {
        if (uid == null || PgSession.role != "resident") {
            onDispose { }
        } else {
            val db = FirebaseFirestore.getInstance()
            val listener = pgQuery(db, "rentRecords")
                .whereEqualTo("residentId", uid)
                .addSnapshotListener { snapshot, _ ->
                    val today = LocalDate.now()
                    val candidates = snapshot?.documents.orEmpty()
                        .mapNotNull { doc ->
                            val due = doc.getString("dueDate") ?: return@mapNotNull null
                            val status = doc.getString("status") ?: "Pending"
                            if (status.equals("Paid", ignoreCase = true)) return@mapNotNull null
                            val dueDate = runCatching { LocalDate.parse(due) }.getOrNull() ?: return@mapNotNull null
                            Triple(doc, dueDate, status)
                        }
                        .sortedBy { it.second }

                    val upcoming = candidates.firstOrNull() ?: return@addSnapshotListener
                    val daysLeft = java.time.temporal.ChronoUnit.DAYS.between(today, upcoming.second)
                    if (daysLeft in 0..3) {
                        val month = upcoming.first.getString("month").orEmpty()
                        val reminderKey = "rent-reminder-$month-$daysLeft"
                        db.collection("notifications")
                            .whereEqualTo("residentId", uid)
                            .get()
                            .addOnSuccessListener { notifications ->
                                val exists = notifications.documents.any {
                                    it.getString("reminderKey") == reminderKey
                                }
                                if (!exists) {
                                    val text = when (daysLeft) {
                                        0L -> "A gentle reminder: your monthly rent is due today. Please complete it when convenient."
                                        1L -> "A gentle reminder: your monthly rent is due tomorrow. Please complete it by the due date."
                                        else -> "A gentle reminder: your monthly rent is due in $daysLeft days. Please complete it by the due date."
                                    }
                                    val data = mapOf(
                                        "residentId" to uid,
                                        "title" to "Rent Reminder",
                                        "message" to text,
                                        "type" to "Rent Reminder",
                                        "reminderKey" to reminderKey,
                                        "isRead" to false,
                                        "createdAt" to System.currentTimeMillis(),
                                        "pgId" to PgSession.pgId
                                    )
                                    db.collection("notifications").add(data)
                                }
                            }
                    }
                }
            onDispose { listener.remove() }
        }
    }
}

// ====================================================
// SERVICE REQUEST SUBMISSION
// ====================================================

fun submitServiceRequest(
    serviceType: String,
    description: String,
    messageSetter: (String) -> Unit
) {
    val user = FirebaseAuth.getInstance().currentUser

    if (user == null) {
        messageSetter("Please log in again before requesting a service.")
        return
    }

    val cleanDescription = description.trim()
    if (cleanDescription.isEmpty()) {
        messageSetter("Please add a short description before submitting the request.")
        return
    }

    val db = FirebaseFirestore.getInstance()
    messageSetter("Submitting $serviceType request...")

    db.collection("residents")
        .document(user.uid)
        .get()
        .addOnSuccessListener { residentDocument ->
            val residentName = residentDocument.getString("fullName")
                ?: user.email
                ?: "Resident"
            val room = residentDocument.getString("room") ?: "Not Assigned"
            val bed = residentDocument.getString("bed") ?: "Not Assigned"

            val requestData = hashMapOf(
                "residentId" to user.uid,
                "residentName" to residentName,
                "residentEmail" to (residentDocument.getString("email") ?: user.email.orEmpty()),
                "room" to room,
                "bed" to bed,
                "serviceType" to serviceType,
                "description" to cleanDescription,
                "status" to "Pending",
                "createdAt" to FieldValue.serverTimestamp(),
                "updatedAt" to FieldValue.serverTimestamp(),
                "pgId" to currentPgId()
            )

            db.collection("serviceRequests")
                .add(withPgId(requestData))
                .addOnSuccessListener {
                    messageSetter("$serviceType request submitted successfully.")
                }
                .addOnFailureListener { error ->
                    messageSetter("Request failed: ${error.message ?: "Unknown error"}")
                }
        }
        .addOnFailureListener { error ->
            messageSetter("Could not load resident details: ${error.message ?: "Unknown error"}")
        }
}

            message = "Please login again to view notifications."
            onDispose { }
        } else {
            val listener = pgQuery(db, "notifications")
                .whereEqualTo("residentId", uid)
                .addSnapshotListener { snapshot, error ->
                    if (error != null) {
                        message =
                            "Could not load notifications: ${error.message ?: "Unknown error"}"
                        return@addSnapshotListener
                    }

                    notifications = snapshot?.documents.orEmpty()
                        .map { document ->
                            ResidentNotification(
                                id = document.id,
                                title = document.getString("title").orEmpty(),
                                body = document.getString("message").orEmpty(),
                                type = document.getString("type") ?: "General",
                                isRead = document.getBoolean("isRead") ?: false,
                                createdAt = document.getLong("createdAt") ?: 0L
                            )
                        }
                        .sortedByDescending { it.createdAt }

                    val unread = notifications.count { !it.isRead }

                    message = if (notifications.isEmpty()) {
                        "No notifications yet."
                    } else {
                        "$unread unread • ${notifications.size} total"
                    }
                }

            onDispose {
                listener.remove()
            }
        }
    }

    Column(
        modifier = Modifier
            .fillMaxSize()

                Brush.verticalGradient(

                    )
                )
            )
            .padding(24.dp)
    ) {
        Row(
            modifier = Modifier.fillMaxWidth(),
            horizontalArrangement = Arrangement.SpaceBetween,
            verticalAlignment = Alignment.CenterVertically
        ) {
            TextButton(onClick = onBack) {
                Text(
                    text = "← Back",
                    color = Color.White
                )
            }

            TextButton(
                onClick = {
                    if (uid != null) {
                        pgQuery(db, "notifications")
                            .whereEqualTo("residentId", uid)
                            .get()
                            .addOnSuccessListener { snapshot ->
                                val batch = db.batch()

                                snapshot.documents.forEach { document ->
                                    batch.update(
                                        document.reference,
                                        "isRead",
                                        true
                                    )
                                }

                                batch.commit()
                            }
                    }
                }
            ) {
                Text(
                    text = "Mark all read",
                    color = Color.White
                )
            }
        }

        Text(
            text = "Notifications",

            fontWeight = FontWeight.Bold,
            color = Color.White
        )

        Spacer(modifier = Modifier.height(8.dp))

        Text(
            text = message,
            fontSize = 14.sp,
            color = Color(0xFFD3E1EE)
        )

        Spacer(modifier = Modifier.height(16.dp))

        if (notifications.isEmpty()) {

@Composable
fun AdminDashboard(
    onFoodManagement: () -> Unit,
    onResidents: () -> Unit,
    onRooms: () -> Unit,
    onComplaints: () -> Unit,
    onNotices: () -> Unit,
    onRentManagement: () -> Unit,
    onServiceRequests: () -> Unit,
    onSearch: () -> Unit,
    onReports: () -> Unit,
    onLogout: () -> Unit
) {
    val db = FirebaseFirestore.getInstance()
    val adminPgId = currentPgId()
    var adminPgName by remember { mutableStateOf(adminPgId) }
    var adminPgNameDraft by remember { mutableStateOf(adminPgId) }
    var pgNameMessage by remember { mutableStateOf("") }

    var totalResidents by remember { mutableStateOf(0) }
    var occupiedRooms by remember { mutableStateOf(0) }
    var pendingComplaints by remember { mutableStateOf(0) }
    var pendingRent by remember { mutableStateOf(0) }
    var pendingServiceRequests by remember { mutableStateOf(0) }
    var totalNotices by remember { mutableStateOf(0) }

    var joinCode by remember { mutableStateOf("Loading...") }
    var joinCodeMessage by remember { mutableStateOf("") }
    var pendingResident by remember { mutableStateOf<PendingResidentSummary?>(null) }
    var pendingResidents by remember { mutableStateOf<List<PendingResidentSummary>>(emptyList()) }
    var rooms by remember { mutableStateOf<List<AdminAssignmentRoom>>(emptyList()) }
    var selectedRoom by remember { mutableStateOf("") }
    var selectedBed by remember { mutableStateOf("") }
    var roomMenuExpanded by remember { mutableStateOf(false) }
    var bedMenuExpanded by remember { mutableStateOf(false) }
    var monthlyRent by remember { mutableStateOf("") }
    var startingDate by remember { mutableStateOf(LocalDate.now().toString()) }
    var rentDueDate by remember { mutableStateOf(LocalDate.now().plusMonths(1).toString()) }
    var approvalMessage by remember { mutableStateOf("") }
    var savingApproval by remember { mutableStateOf(false) }

    DisposableEffect(adminPgId) {
        val pgListener = db.collection("pgs").document(adminPgId)
            .addSnapshotListener { snapshot, _ ->
                if (snapshot != null && snapshot.exists()) {
                    adminPgName = snapshot.getString("name")
                        ?: snapshot.getString("pgName")
                                ?: snapshot.getString("displayName")
                                ?: adminPgId
                    if (pgNameMessage.isBlank()) adminPgNameDraft = adminPgName
                } else {
                    adminPgName = adminPgId
                }
            }
        val residentListener = db.collection("residents")
            .whereEqualTo("pgId", adminPgId)
            .addSnapshotListener { snapshot, _ ->
                if (snapshot != null) {
                    totalResidents = snapshot.documents.count { it.getString("status") == "Active" }
                    occupiedRooms = snapshot.documents
                        .filter { it.getString("status") == "Active" }
                        .mapNotNull { it.getString("room")?.trim() }
                        .filter { it.isNotEmpty() && it != "Not Assigned" }
                        .toSet()
                        .size
                }
            }

        val pendingListener = db.collection("residents")
            .whereEqualTo("requestedPgId", adminPgId)
            .whereEqualTo("status", "Pending")
            .addSnapshotListener { snapshot, error ->
                if (error == null) {
                    pendingResidents = snapshot?.documents?.map { doc ->
                        PendingResidentSummary(
                            id = doc.id,
                            fullName = doc.getString("fullName").orEmpty(),
                            email = doc.getString("email").orEmpty(),
                            phone = doc.getString("phone").orEmpty(),
                            requestedPgId = doc.getString("requestedPgId").orEmpty()
                        )
                    }?.sortedBy { it.fullName.lowercase() } ?: emptyList()

                    if (pendingResident == null && pendingResidents.isNotEmpty()) {
                        pendingResident = pendingResidents.first()
                    }
                }
            }

        val roomListener = db.collection("rooms")
            .whereEqualTo("pgId", adminPgId)
            .addSnapshotListener { snapshot, _ ->
                rooms = snapshot?.documents?.map { doc ->
                    AdminAssignmentRoom(
                        id = doc.id,
                        roomNumber = doc.getString("roomNumber") ?: doc.id,
                        roomType = doc.getString("roomType") ?: "Shared",
                        floor = doc.getString("floor") ?: "-",
                        totalBeds = (doc.getLong("totalBeds") ?: 2L).toInt(),
                        occupiedBeds = (doc.getLong("occupiedBeds") ?: 0L).toInt()
                    )
                }?.sortedBy { it.roomNumber } ?: emptyList()
            }

        val complaintListener = pgQuery(db, "complaints")
            .addSnapshotListener { snapshot, _ ->
                pendingComplaints = snapshot?.documents?.count {
                    (it.getString("status") ?: "Pending") == "Pending"
                } ?: 0
            }

        val rentListener = pgQuery(db, "rentRecords")
            .addSnapshotListener { snapshot, _ ->
                pendingRent = snapshot?.documents?.count {
                    (it.getString("status") ?: "Pending") == "Pending"
                } ?: 0
            }

        val serviceListener = pgQuery(db, "serviceRequests")
            .addSnapshotListener { snapshot, _ ->
                pendingServiceRequests = snapshot?.documents?.count {
                    (it.getString("status") ?: "Pending") == "Pending"
                } ?: 0
            }

        val noticeListener = pgQuery(db, "notices")
            .addSnapshotListener { snapshot, _ ->
                totalNotices = snapshot?.size() ?: 0
            }

        val joinCodeListener = db.collection("pgJoinCodes")
            .whereEqualTo("pgId", adminPgId)
            .whereEqualTo("active", true)
            .limit(1)
            .addSnapshotListener { snapshot, error ->
                if (error == null) {
                    joinCode = snapshot?.documents?.firstOrNull()?.id ?: "Not generated"
                } else {
                    joinCode = "Unavailable"
                }
            }

        onDispose {
            pgListener.remove()
            residentListener.remove()
            pendingListener.remove()
            roomListener.remove()
            complaintListener.remove()
            rentListener.remove()
            serviceListener.remove()
            noticeListener.remove()
            joinCodeListener.remove()
        }
    }

    fun openPendingResident(resident: PendingResidentSummary) {
        pendingResident = resident
        selectedRoom = ""
        selectedBed = ""
        monthlyRent = ""
        startingDate = LocalDate.now().toString()
        rentDueDate = LocalDate.now().plusMonths(1).toString()
        approvalMessage = ""
    }

    fun generateJoinCode() {
        joinCodeMessage = "Generating join code..."
        val newCode = "PG-${UUID.randomUUID().toString().replace("-", "").take(8).uppercase(Locale.getDefault())}"
        val codeRef = db.collection("pgJoinCodes").document(newCode)
        codeRef.get().addOnSuccessListener { existing ->
            if (existing.exists()) {
                generateJoinCode()
            } else {
                db.collection("pgJoinCodes")
                    .whereEqualTo("pgId", adminPgId)
                    .whereEqualTo("active", true)
                    .get()
                    .addOnSuccessListener { oldCodes ->
                        val batch = db.batch()
                        oldCodes.documents.forEach { batch.update(it.reference, "active", false) }
                        batch.set(codeRef, mapOf(
                            "pgId" to adminPgId,
                            "createdBy" to (FirebaseAuth.getInstance().currentUser?.uid ?: ""),
                            "active" to true,
                            "createdAt" to FieldValue.serverTimestamp()
                        ))
                        batch.commit()
                            .addOnSuccessListener { joinCodeMessage = "New join code created." }
                            .addOnFailureListener { error -> joinCodeMessage = "Could not create code: ${error.message ?: "Unknown error"}" }
                    }
                    .addOnFailureListener { error -> joinCodeMessage = "Could not prepare code: ${error.message ?: "Unknown error"}" }
            }
        }.addOnFailureListener { error -> joinCodeMessage = "Could not create code: ${error.message ?: "Unknown error"}" }
    }

    fun approvePendingResident() {
        val resident = pendingResident ?: return
        val roomNumber = selectedRoom.trim()
        val bed = selectedBed.trim()
        val rent = monthlyRent.trim().toDoubleOrNull()
        val start = try { LocalDate.parse(startingDate.trim()) } catch (_: Exception) { null }
        val due = try { LocalDate.parse(rentDueDate.trim()) } catch (_: Exception) { null }

        if (roomNumber.isBlank() || bed.isBlank()) {
            approvalMessage = "Select/enter a room and bed."
            return
        }
        if (rent == null || rent <= 0) {
            approvalMessage = "Enter a valid monthly rent."
            return
        }
        if (start == null || due == null) {
            approvalMessage = "Use dates in YYYY-MM-DD format."
            return
        }
        if (due.isBefore(start)) {
            approvalMessage = "Rent due date cannot be before the starting date."
            return
        }

        val room = rooms.firstOrNull { it.roomNumber.equals(roomNumber, ignoreCase = true) }
        if (room == null) {
            approvalMessage = "Room $roomNumber does not exist in $adminPgId."
            return
        }

        savingApproval = true
        approvalMessage = "Checking room and saving approval..."

        db.collection("residents")
            .whereEqualTo("pgId", adminPgId)
            .whereEqualTo("room", roomNumber)
            .whereEqualTo("bed", bed)
            .whereEqualTo("status", "Active")
            .get()
            .addOnSuccessListener { occupied ->
                if (occupied.documents.any { it.id != resident.id }) {
                    savingApproval = false
                    approvalMessage = "Bed $bed in room $roomNumber is already occupied."
                    return@addOnSuccessListener
                }

                val residentRef = db.collection("residents").document(resident.id)
                val updateData = mapOf(
                    "pgId" to adminPgId,
                    "requestedPgId" to adminPgId,
                    "status" to "Active",
                    "approved" to true,
                    "approvedBy" to (FirebaseAuth.getInstance().currentUser?.uid ?: ""),
                    "room" to roomNumber,
                    "bed" to bed,
                    "roomType" to room.roomType,
                    "floor" to room.floor,
                    "monthlyRent" to rent,
                    "rentDueDay" to due.dayOfMonth,
                    "startingDate" to start.toString(),
                    "rentDueDate" to due.toString(),
                    "updatedAt" to FieldValue.serverTimestamp()
                )

                residentRef.update(updateData)
                    .addOnSuccessListener {
                        val month = due.format(DateTimeFormatter.ofPattern("MMMM yyyy", Locale.getDefault()))
                        val rentData = mapOf(
                            "residentId" to resident.id,
                            "residentName" to resident.fullName,
                            "residentEmail" to resident.email,
                            "room" to roomNumber,
                            "bed" to bed,
                            "month" to month,
                            "amount" to rent.toString(),
                            "dueDate" to due.toString(),
                            "status" to "Pending",
                            "startingDate" to start.toString(),
                            "createdAt" to FieldValue.serverTimestamp(),
                            "updatedAt" to FieldValue.serverTimestamp(),
                            "pgId" to adminPgId
                        )

                        db.collection("rentRecords")
                            .whereEqualTo("pgId", adminPgId)
                            .whereEqualTo("residentId", resident.id)
                            .whereEqualTo("month", month)
                            .get()
                            .addOnSuccessListener { existing ->
                                val finishApproval: () -> Unit = {
                                    createResidentNotification(
                                        resident.id,
                                        "Account Approved",
                                        "Your ResidentHub account is active. PG $adminPgId, Room $roomNumber, Bed $bed. Monthly rent: ₹${rent.toString()}.",
                                        "Account"
                                    )
                                    savingApproval = false
                                    approvalMessage = "${resident.fullName} approved successfully."
                                    pendingResidents = pendingResidents.filterNot { it.id == resident.id }
                                    pendingResident = pendingResidents.firstOrNull()
                                }

                                if (existing.isEmpty()) {
                                    db.collection("rentRecords").add(rentData)
                                        .addOnSuccessListener { finishApproval() }
                                        .addOnFailureListener { error ->
                                            savingApproval = false
                                            approvalMessage = "Resident approved, but rent record failed: ${error.message ?: "Unknown error"}"
                                        }
                                } else {
                                    finishApproval()
                                }
                            }
                            .addOnFailureListener { error ->
                                savingApproval = false
                                approvalMessage = "Resident approved, but rent check failed: ${error.message ?: "Unknown error"}"
                            }
                    }
                    .addOnFailureListener { error ->
                        savingApproval = false
                        approvalMessage = "Approval failed: ${error.message ?: "Unknown error"}"
                    }
            }
            .addOnFailureListener { error ->
                savingApproval = false
                approvalMessage = "Could not check bed availability: ${error.message ?: "Unknown error"}"
            }
    }

    }

    if (pendingResident != null) {
        val resident = pendingResident!!
        AlertDialog(
            onDismissRequest = {
                if (!savingApproval) pendingResident = null
            },
            title = { Text("New Resident Request") },
            text = {

@Composable
fun AdminResidentsScreen(
    onBack: () -> Unit
) {
    val db = FirebaseFirestore.getInstance()
    val adminPgId = currentPgId()

    var residents by remember { mutableStateOf<List<ResidentItem>>(emptyList()) }
    var rooms by remember { mutableStateOf<List<ResidentRoomOption>>(emptyList()) }
    var search by remember { mutableStateOf("") }
    var message by remember { mutableStateOf("Loading residents...") }
    var selectedResident by remember { mutableStateOf<ResidentItem?>(null) }
    var selectedRoom by remember { mutableStateOf("") }
    var selectedBed by remember { mutableStateOf("") }
    var roomMenuExpanded by remember { mutableStateOf(false) }
    var bedMenuExpanded by remember { mutableStateOf(false) }
    var monthlyRent by remember { mutableStateOf("") }
    var rentDueDay by remember { mutableStateOf("10") }
    var saving by remember { mutableStateOf(false) }

    DisposableEffect(Unit) {
        fun mapResident(document: com.google.firebase.firestore.DocumentSnapshot): ResidentItem {
            return ResidentItem(
                id = document.id,
                fullName = document.getString("fullName").orEmpty(),
                email = document.getString("email").orEmpty(),
                phone = document.getString("phone").orEmpty(),
                room = document.getString("room") ?: "Not Assigned",
                status = document.getString("status") ?: "Pending",
                bed = document.getString("bed") ?: "Not Assigned",
                roomType = document.getString("roomType") ?: "Shared",
                floor = document.getString("floor") ?: "-",
                monthlyRent = firestoreNumber(document.get("monthlyRent")),
                rentDueDay = (document.getLong("rentDueDay") ?: 0L).toInt()
            )
        }

        var assignedResidents = emptyList<ResidentItem>()
        var pendingResidents = emptyList<ResidentItem>()

        fun publishResidents() {
            residents = (pendingResidents + assignedResidents)
                .distinctBy { it.id }
                .sortedWith(compareBy<ResidentItem> { it.status != "Pending" }.thenBy { it.fullName.lowercase() })
            val pendingCount = pendingResidents.size
            message = if (pendingCount > 0) "$pendingCount pending approval(s). PG: $adminPgId" else "PG $adminPgId resident management ready."
        }

        val assignedListener = db.collection("residents")
            .whereEqualTo("pgId", adminPgId)
            .addSnapshotListener { snapshot, error ->
                if (error != null) {
                    message = "Could not load assigned residents: ${error.message ?: "Unknown error"}"
                } else {
                    assignedResidents = snapshot?.documents?.map(::mapResident) ?: emptyList()
                    publishResidents()
                }
            }

        val pendingListener = db.collection("residents")
            .whereEqualTo("requestedPgId", adminPgId)
            .whereEqualTo("status", "Pending")
            .addSnapshotListener { snapshot, error ->
                if (error != null) {
                    message = "Could not load pending residents: ${error.message ?: "Unknown error"}"
                } else {
                    pendingResidents = snapshot?.documents?.map(::mapResident) ?: emptyList()
                    publishResidents()
                }
            }

        val roomListener = db.collection("rooms")
            .whereEqualTo("pgId", adminPgId)
            .addSnapshotListener { snapshot, error ->
                if (error == null) {
                    rooms = snapshot?.documents?.map { document ->
                        ResidentRoomOption(
                            roomNumber = document.getString("roomNumber") ?: document.id,
                            roomType = document.getString("roomType") ?: "Shared",
                            floor = document.getString("floor") ?: "-",
                            totalBeds = (document.getLong("totalBeds") ?: 2L).toInt()
                        )
                    }?.sortedBy { it.roomNumber } ?: emptyList()
                }
            }

        onDispose {
            assignedListener.remove()
            pendingListener.remove()
            roomListener.remove()
        }
    }

    fun startApproval(resident: ResidentItem) {
        selectedResident = resident
        selectedRoom = if (resident.room == "Not Assigned") "" else resident.room
        selectedBed = if (resident.bed == "Not Assigned") "" else resident.bed
        roomMenuExpanded = false
        bedMenuExpanded = false
        monthlyRent = ""
        rentDueDay = "10"
        message = "Assign PG, room, bed and rent for ${resident.fullName}."
    }

    fun approveResident() {
        val resident = selectedResident ?: return
        val roomNumber = selectedRoom.trim()
        val bed = selectedBed.trim()
        val rent = monthlyRent.trim().toDoubleOrNull()
        val dueDay = rentDueDay.trim().toIntOrNull()

        if (roomNumber.isBlank() || bed.isBlank()) {
            message = "PG, room and bed assignment are required."
            return
        }
        if (rent == null || rent <= 0) {
            message = "Enter a valid monthly rent."
            return
        }
        if (dueDay == null || dueDay !in 1..28) {
            message = "Rent due day must be between 1 and 28."
            return
        }

        val roomDoc = rooms.firstOrNull { it.roomNumber.equals(roomNumber, true) }
        if (roomDoc == null) {
            message = "Room $roomNumber does not exist in PG $adminPgId."
            return
        }

        saving = true
        db.collection("residents")
            .whereEqualTo("pgId", adminPgId)
            .whereEqualTo("room", roomNumber)
            .whereEqualTo("bed", bed)
            .get()
            .addOnSuccessListener { occupiedSnapshot ->
                val occupiedByOther = occupiedSnapshot.documents.any { it.id != resident.id && (it.getString("status") ?: "Active") == "Active" }
                if (occupiedByOther) {
                    saving = false
                    message = "Bed $bed in room $roomNumber is already occupied."
                    return@addOnSuccessListener
                }

                val residentRef = db.collection("residents").document(resident.id)
                val updateData = mapOf(
                    "pgId" to adminPgId,
                    "status" to "Active",
                    "approved" to true,
                    "approvedBy" to (FirebaseAuth.getInstance().currentUser?.uid ?: ""),
                    "room" to roomNumber,
                    "bed" to bed,
                    "roomType" to roomDoc.roomType,
                    "floor" to roomDoc.floor,
                    "monthlyRent" to rent,
                    "rentDueDay" to dueDay,
                    "updatedAt" to FieldValue.serverTimestamp()
                )

                residentRef.update(updateData)
                    .addOnSuccessListener {
                        val now = LocalDate.now()
                        val month = now.format(DateTimeFormatter.ofPattern("MMMM yyyy", Locale.getDefault()))
                        val dueDate = LocalDate.of(now.year, now.month, dueDay).toString()
                        val rentData = mapOf(
                            "residentId" to resident.id,
                            "residentName" to resident.fullName,
                            "residentEmail" to resident.email,
                            "room" to roomNumber,
                            "bed" to bed,
                            "month" to month,
                            "amount" to rent.toString(),
                            "dueDate" to dueDate,
                            "status" to "Pending",
                            "createdAt" to FieldValue.serverTimestamp(),
                            "updatedAt" to FieldValue.serverTimestamp(),
                            "pgId" to adminPgId
                        )

                        db.collection("rentRecords")
                            .whereEqualTo("pgId", adminPgId)
                            .whereEqualTo("residentId", resident.id)
                            .whereEqualTo("month", month)
                            .get()
                            .addOnSuccessListener { existing ->
                                val finish: () -> Unit = {
                                    createResidentNotification(
                                        resident.id,
                                        "Account Approved",
                                        "Your account is approved for PG $adminPgId. Room $roomNumber, Bed $bed. Monthly rent: ₹${rent.toString()}.",
                                        "Account"
                                    )
                                    saving = false
                                    message = "${resident.fullName} approved and assigned to PG $adminPgId, room $roomNumber, bed $bed."
                                    selectedResident = null
                                }

                                if (existing.isEmpty()) {
                                    db.collection("rentRecords").add(rentData)
                                        .addOnSuccessListener { finish() }
                                        .addOnFailureListener { error -> saving = false; message = "Resident approved, but rent record failed: ${error.message ?: "Unknown error"}" }
                                } else {
                                    finish()
                                }
                            }
                            .addOnFailureListener { error -> saving = false; message = "Resident approved, but rent check failed: ${error.message ?: "Unknown error"}" }
                    }
                    .addOnFailureListener { error -> saving = false; message = "Approval failed: ${error.message ?: "Unknown error"}" }
            }
            .addOnFailureListener { error -> saving = false; message = "Could not check bed availability: ${error.message ?: "Unknown error"}" }
    }

    val filteredResidents = residents.filter {
        val q = search.trim().lowercase()
        q.isEmpty() || it.fullName.lowercase().contains(q) || it.email.lowercase().contains(q) || it.phone.contains(q) || it.room.lowercase().contains(q) || it.bed.lowercase().contains(q) || it.status.lowercase().contains(q)
    }
    val pending = filteredResidents.filter { it.status == "Pending" && it.room == "Not Assigned" }
    val active = filteredResidents.filter { !(it.status == "Pending" && it.room == "Not Assigned") }

    Column(modifier = Modifier.fillMaxSize().padding(20.dp)) {
        TextButton(onClick = onBack) { Text("← Back") }
        Text("Resident Management", fontSize = 28.sp, fontWeight = FontWeight.Bold)
        Text("Admin PG: $adminPgId", fontSize = 14.sp)
        Spacer(modifier = Modifier.height(6.dp))
        Text(message, fontSize = 14.sp)
        Spacer(modifier = Modifier.height(12.dp))

        if (selectedResident != null) {
            Card(modifier = Modifier.fillMaxWidth(), elevation = CardDefaults.cardElevation(defaultElevation = 4.dp)) {
                Column(modifier = Modifier.padding(16.dp)) {
                    Text("Approve & Assign Resident", fontSize = 21.sp, fontWeight = FontWeight.Bold)
                    Spacer(modifier = Modifier.height(6.dp))
                    Text("Resident: ${selectedResident!!.fullName}")
                    Text("Email: ${selectedResident!!.email}")
                    Spacer(modifier = Modifier.height(10.dp))
                    Text("PG: $adminPgId", fontWeight = FontWeight.Bold)
                    Spacer(modifier = Modifier.height(8.dp))
                    val currentResidentId = selectedResident!!.id
                    val selectedRoomOption = rooms.firstOrNull { it.roomNumber.equals(selectedRoom, ignoreCase = true) }
                    val occupiedBedsInSelectedRoom = residents
                        .filter { it.id != currentResidentId && it.status.equals("Active", ignoreCase = true) && it.room.equals(selectedRoom, ignoreCase = true) }
                        .map { it.bed.trim() }
                        .filter { it.isNotBlank() && !it.equals("Not Assigned", ignoreCase = true) }
                    val normalizedOccupiedBeds = occupiedBedsInSelectedRoom.map { bed ->
                        bed.removePrefix("Bed ").trim().lowercase()
                    }.toSet()
                    val availableBedCount = selectedRoomOption?.let { option ->
                        (1..option.totalBeds).count { bedNumber ->
                            bedNumber.toString().lowercase() !in normalizedOccupiedBeds
                        }
                    } ?: 0

                    Box(modifier = Modifier.fillMaxWidth()) {
                        OutlinedTextField(
                            value = selectedRoom,
                            onValueChange = {},
                            readOnly = true,
                            label = { Text("Room Number") },
                            supportingText = {
                                if (selectedRoomOption != null) {
                                    Text("Occupied: ${occupiedBedsInSelectedRoom.size}/${selectedRoomOption.totalBeds} • Available: $availableBedCount")
                                } else {
                                    Text("Select a room")
                                }
                            },
                            modifier = Modifier.fillMaxWidth().clickable { roomMenuExpanded = true }
                        )
                        DropdownMenu(
                            expanded = roomMenuExpanded,
                            onDismissRequest = { roomMenuExpanded = false },
                            modifier = Modifier.fillMaxWidth()
                        ) {
                            rooms.forEach { room ->
                                val occupiedCount = residents.count {
                                    it.id != currentResidentId &&
                                            it.status.equals("Active", ignoreCase = true) &&
                                            it.room.equals(room.roomNumber, ignoreCase = true) &&
                                            it.bed.isNotBlank() &&
                                            !it.bed.equals("Not Assigned", ignoreCase = true)
                                }
                                val availableCount = (room.totalBeds - occupiedCount).coerceAtLeast(0)
                                DropdownMenuItem(
                                    text = {
                                        Text("Room ${room.roomNumber} • ${room.roomType} • $occupiedCount/${room.totalBeds} occupied • $availableCount available")
                                    },
                                    enabled = availableCount > 0,
                                    onClick = {
                                        selectedRoom = room.roomNumber
                                        selectedBed = ""
                                        roomMenuExpanded = false
                                        bedMenuExpanded = false
                                    }
                                )
                            }
                        }
                    }
                    Spacer(modifier = Modifier.height(8.dp))
                    Box(modifier = Modifier.fillMaxWidth()) {
                        OutlinedTextField(
                            value = selectedBed,
                            onValueChange = {},
                            readOnly = true,
                            label = { Text("Bed") },
                            supportingText = {
                                if (selectedRoomOption != null) {
                                    Text("${selectedRoomOption.totalBeds} total • ${occupiedBedsInSelectedRoom.size} reserved • $availableBedCount available")
                                } else {
                                    Text("Select a room first")
                                }
                            },
                            modifier = Modifier.fillMaxWidth().clickable {
                                if (selectedRoomOption != null) bedMenuExpanded = true
                            }
                        )
                        DropdownMenu(
                            expanded = bedMenuExpanded,
                            onDismissRequest = { bedMenuExpanded = false },
                            modifier = Modifier.fillMaxWidth()
                        ) {
                            if (selectedRoomOption != null) {
                                (1..selectedRoomOption.totalBeds).forEach { bedNumber ->
                                    val bedName = "Bed $bedNumber"
                                    val occupied = normalizedOccupiedBeds.contains(bedNumber.toString().lowercase())
                                    DropdownMenuItem(
                                        text = { Text(if (occupied) "$bedName • Occupied" else "$bedName • Available") },
                                        enabled = !occupied,
                                        onClick = {
                                            selectedBed = bedName
                                            bedMenuExpanded = false
                                        }
                                    )
                                }
                            }
                        }
                    }
                    Spacer(modifier = Modifier.height(8.dp))
                    OutlinedTextField(value = monthlyRent, onValueChange = { monthlyRent = it }, label = { Text("Monthly Rent ₹") }, modifier = Modifier.fillMaxWidth())
                    Spacer(modifier = Modifier.height(8.dp))
                    OutlinedTextField(value = rentDueDay, onValueChange = { rentDueDay = it }, label = { Text("Rent Due Day (1-28)") }, modifier = Modifier.fillMaxWidth())
                    Spacer(modifier = Modifier.height(8.dp))
                    if (rooms.isNotEmpty()) Text("Available rooms: ${rooms.joinToString { it.roomNumber }}", fontSize = 12.sp)
                    Spacer(modifier = Modifier.height(10.dp))
                    Button(enabled = !saving, onClick = { approveResident() }, modifier = Modifier.fillMaxWidth()) { Text(if (saving) "Assigning..." else "Approve & Assign") }
                    TextButton(enabled = !saving, onClick = { selectedResident = null }, modifier = Modifier.fillMaxWidth()) { Text("Cancel") }
                }
            }
            Spacer(modifier = Modifier.height(16.dp))
        }

        OutlinedTextField(value = search, onValueChange = { search = it }, label = { Text("Search residents") }, modifier = Modifier.fillMaxWidth())
        Spacer(modifier = Modifier.height(16.dp))

        Text("Pending Approval (${pending.size})", fontSize = 21.sp, fontWeight = FontWeight.Bold)
        Spacer(modifier = Modifier.height(8.dp))
        LazyColumn(verticalArrangement = Arrangement.spacedBy(10.dp), modifier = Modifier.weight(1f)) {
            if (pending.isEmpty()) {
                item { Text("No pending residents.") }
            } else {
                items(pending) { resident ->
                    Card(modifier = Modifier.fillMaxWidth(), elevation = CardDefaults.cardElevation(defaultElevation = 3.dp)) {
                        Column(modifier = Modifier.padding(14.dp)) {
                            Text(resident.fullName, fontSize = 18.sp, fontWeight = FontWeight.Bold)
                            Text(resident.email)
                            Text("Phone: ${resident.phone}")
                            Text("Status: Pending")
                            Spacer(modifier = Modifier.height(8.dp))
                            Button(onClick = { startApproval(resident) }, modifier = Modifier.fillMaxWidth()) { Text("Review & Assign PG + Room + Bed + Rent") }
                        }
                    }
                }
            }
            item { Spacer(modifier = Modifier.height(8.dp)); Text("Active / Existing Residents", fontSize = 21.sp, fontWeight = FontWeight.Bold) }
            items(active) { resident ->
                AdminResidentCard(
                    resident = resident,
                    onEdit = { startApproval(resident) },
                    onDeactivate = {
                        db.collection("residents").document(resident.id).update(
                            mapOf("status" to "Inactive", "approved" to false, "room" to "Not Assigned", "bed" to "Not Assigned", "updatedAt" to FieldValue.serverTimestamp())
                        ).addOnSuccessListener { message = "${resident.fullName} deactivated." }
                            .addOnFailureListener { error -> message = "Deactivate failed: ${error.message ?: "Unknown error"}" }
                    },
                    onDelete = {
                        db.collection("residents").document(resident.id).delete()
                            .addOnSuccessListener { message = "Resident profile deleted." }
                            .addOnFailureListener { error -> message = "Delete failed: ${error.message ?: "Unknown error"}" }
                    }
                )
            }
        }
    }
}

@Composable
fun AdminRoomsScreen(
    onBack: () -> Unit
) {
    val db = FirebaseFirestore.getInstance()

    var residents by remember { mutableStateOf<List<ResidentRoomItem>>(emptyList()) }
    var rooms by remember { mutableStateOf<List<RoomInventoryItem>>(emptyList()) }
    var message by remember { mutableStateOf("Loading room data...") }

    var newRoomNumber by remember { mutableStateOf("") }
    var newFloor by remember { mutableStateOf("") }
    var newRoomType by remember { mutableStateOf("Shared") }
    var newTotalBeds by remember { mutableStateOf("2") }

    DisposableEffect(Unit) {
        val residentListener = pgQuery(db, "residents")
            .addSnapshotListener { snapshot, error ->
                if (error != null) {
                    message = "Could not load residents: ${error.message ?: "Unknown error"}"
                    return@addSnapshotListener
                }

                residents = snapshot?.documents?.map { document ->
                    ResidentRoomItem(
                        id = document.id,
                        fullName = document.getString("fullName").orEmpty(),
                        email = document.getString("email").orEmpty(),
                        room = document.getString("room") ?: "Not Assigned",
                        bed = document.getString("bed") ?: "Not Assigned",
                        roomType = document.getString("roomType") ?: "Shared",
                        floor = document.getString("floor") ?: "-"
                    )
                } ?: emptyList()
            }

        val roomListener = pgQuery(db, "rooms")
            .addSnapshotListener { snapshot, error ->
                if (error != null) {
                    message = "Could not load rooms: ${error.message ?: "Unknown error"}"
                    return@addSnapshotListener
                }

                rooms = snapshot?.documents?.map { document ->
                    RoomInventoryItem(
                        id = document.id,

data class RoomInventoryItem(
    val id: String,
    val roomNumber: String,
    val floor: String,
    val roomType: String,
    val totalBeds: Int
)

@Composable
fun RoomAvailabilityCard(
    room: RoomInventoryItem,
    occupiedBeds: Int
) {
    val safeOccupied = occupiedBeds.coerceAtMost(room.totalBeds)
    val availableBeds = (room.totalBeds - safeOccupied).coerceAtLeast(0)
    val status = when {
        safeOccupied == 0 -> "Available"
        safeOccupied >= room.totalBeds -> "Full"
        else -> "Partially Occupied"
    }

    Card(
        modifier = Modifier.fillMaxWidth(),
        elevation = CardDefaults.cardElevation(defaultElevation = 3.dp)
    ) {
        Column(modifier = Modifier.padding(16.dp)) {
            Text(
                "Room ${room.roomNumber}",
                fontSize = 19.sp,
                fontWeight = FontWeight.Bold
            )
            Text("Floor: ${room.floor}")
            Text("Type: ${room.roomType}")
            Text("Total Beds: ${room.totalBeds}")
            Text("Occupied: $safeOccupied")
            Text("Available: $availableBeds")

            Spacer(modifier = Modifier.height(8.dp))

            Text(
                text = "Status: $status",
                fontWeight = FontWeight.Bold
            )
        }
    }
}

@Composable
fun AdminComplaintsScreen(

) {

    var message by remember { mutableStateOf("Loading complaints...") }

        val listener = pgQuery(db, "complaints")
            .orderBy("createdAt", com.google.firebase.firestore.Query.Direction.DESCENDING)
            .addSnapshotListener { snapshot, error ->
                if (error != null) {
                    message = "Could not load complaints: ${error.message ?: "Unknown error"}"
                    return@addSnapshotListener
                }

                if (snapshot == null) {
                    complaints = emptyList()
                    message = "No complaints found."
                    return@addSnapshotListener
                }

                complaints = snapshot.documents.map { document ->
                    ComplaintItem(
                        id = document.id,
                        title = document.getString("title").orEmpty(),
                        category = document.getString("category").orEmpty(),
                        description = document.getString("description").orEmpty(),
                        status = document.getString("status") ?: "Pending"
                    )
                }

                message = if (complaints.isEmpty()) {
                    "No complaints submitted yet."
                } else {
                    "${complaints.size} complaint(s) found."
                }
            }

        onDispose {
            listener.remove()
        }
    }

    Column(
        modifier = Modifier
            .fillMaxSize()
            .padding(24.dp)
    ) {
        TextButton(onClick = onBack) {
            Text("← Back")
        }

        Text(
            text = "Complaints",
            fontSize = 28.sp,
            fontWeight = FontWeight.Bold
        )

        Spacer(modifier = Modifier.height(8.dp))

        Text(
            text = message,
            fontSize = 14.sp
        )

        Spacer(modifier = Modifier.height(16.dp))

        if (complaints.isEmpty()) {
            Text("No complaints to display.")
        } else {
            androidx.compose.foundation.lazy.LazyColumn(
                verticalArrangement = Arrangement.spacedBy(12.dp),
                modifier = Modifier.fillMaxSize()
            ) {
                items(complaints) { complaint ->
                    AdminComplaintCard(
                        complaint = complaint,
                        onStatusChange = { newStatus ->
                            db.collection("complaints")
                                .document(complaint.id)
                                .update("status", newStatus)
                                .addOnSuccessListener {
                                    db.collection("complaints")
                                        .document(complaint.id)
                                        .get()
                                        .addOnSuccessListener { document ->
                                            val residentId = document.getString("residentId").orEmpty()
                                            if (residentId.isNotBlank()) {
                                                createResidentNotification(
                                                    residentId = residentId,
                                                    title = "Complaint Update",
                                                    message = "${complaint.title} is now $newStatus.",
                                                    type = "Complaint"
                                                )
                                            }
                                        }
                                }
                                .addOnFailureListener { error ->
                                    message =
                                        "Status update failed: ${error.message ?: "Unknown error"}"
                                }
                        }
                    )
                }
            }
        }
    }
}

    var selectedFilter by remember { mutableStateOf("All") }
    var message by remember { mutableStateOf("") }
    var notices by remember { mutableStateOf<List<AdminNoticeItem>>(emptyList()) }

        val listener = pgQuery(db, "notices")
            .addSnapshotListener { snapshot, error ->
                if (error != null) {
                    message = "Could not load notices: ${error.message ?: "Unknown error"}"
                    return@addSnapshotListener
                }

                    AdminNoticeItem(
                        id = document.id,
                        title = document.getString("title").orEmpty(),
                        date = document.getString("date").orEmpty(),
                        description = document.getString("description").orEmpty(),
                        type = document.getString("type") ?: "General",

                }.sortedByDescending { it.createdAt }
            }

        onDispose { listener.remove() }

        val matchesSearch = search.isBlank() ||
                notice.title.contains(search, ignoreCase = true) ||
                notice.description.contains(search, ignoreCase = true) ||
                notice.type.contains(search, ignoreCase = true)
        val matchesFilter = selectedFilter == "All" || notice.type.equals(selectedFilter, ignoreCase = true)
        matchesSearch && matchesFilter
    }

    Column(
        modifier = Modifier
            .fillMaxSize()
            .padding(24.dp)
    ) {
        TextButton(onClick = onBack) { Text("← Back") }

        Text(
            text = if (editingId == null) "Notice Publishing" else "Edit Notice",

            label = { Text("Category") },

            value = description,
            onValueChange = { description = it },
            label = { Text("Description") },

                .fillMaxWidth()
                .height(120.dp)
        )
        Spacer(modifier = Modifier.height(12.dp))

        Button(

                if (title.isBlank() || date.isBlank() || description.isBlank()) {
                    message = "Please fill in title, date, and description."
                    return@Button
                }

                val data = hashMapOf(
                    "title" to title.trim(),
                    "date" to date.trim(),
                    "description" to description.trim(),
                    "type" to type.trim().ifBlank { "General" },
                    "createdAt" to System.currentTimeMillis()
                )

                val currentEditingId = editingId
                val wasEditing = currentEditingId != null

                fun clearForm() {
                    editingId = null
                    title = ""
                    date = ""
                    description = ""
                    type = "General"

                if (currentEditingId == null) {

                            pgQuery(db, "residents").get()

                                            residentId = resident.id,

                        }
                } else {
                    db.collection("notices").document(currentEditingId)
                        .set(data, SetOptions.merge())
                        .addOnSuccessListener {
                            message = "Notice updated successfully."
                            clearForm()
                        }
                        .addOnFailureListener { error ->
                            message = "Update failed: ${error.message ?: "Unknown error"}"
                        }
                }

        }

                editingId = null
                title = ""
                date = ""
                description = ""
                type = "General"
                message = "Edit cancelled."
            }) {
                Text("Cancel Edit")
            }
        }

        Spacer(modifier = Modifier.height(8.dp))
        Text(message, fontSize = 14.sp)
        Spacer(modifier = Modifier.height(16.dp))

        Text("Published Notices", fontSize = 20.sp, fontWeight = FontWeight.Bold)
        Spacer(modifier = Modifier.height(8.dp))

        OutlinedTextField(
            value = search,
            onValueChange = { search = it },
            label = { Text("Search notices") },
            modifier = Modifier.fillMaxWidth()

                    label = { Text(filter) }
                )
            }
        }
        Spacer(modifier = Modifier.height(10.dp))

        if (filteredNotices.isEmpty()) {
            Text("No notices match your search/filter.")
        } else {

                verticalArrangement = Arrangement.spacedBy(12.dp),
                modifier = Modifier.fillMaxSize()
            ) {

                    Card(
                        modifier = Modifier.fillMaxWidth(),

                                        .addOnSuccessListener { message = "Notice deleted." }
                                        .addOnFailureListener { error ->
                                            message = "Delete failed: ${error.message ?: "Unknown error"}"
                                        }
                                }) { Text("Delete") }
                            }
                        }
                    }
                }
            }
        }
    }
}

@Composable
fun AdminRentCard(
    resident: ResidentItem,
    existingRecords: List<AdminRentRecord>,
    db: FirebaseFirestore,
    onMessage: (String) -> Unit
) {
    var month by remember(resident.id) { mutableStateOf("October 2026") }
    var amount by remember(resident.id) { mutableStateOf("12000") }
    var dueDate by remember(resident.id) { mutableStateOf("10 October 2026") }
    var status by remember(resident.id) { mutableStateOf("Pending") }

        modifier = Modifier.fillMaxWidth(),
        elevation = CardDefaults.cardElevation(defaultElevation = 3.dp)
    ) {

                modifier = Modifier.fillMaxWidth()
            )

            Spacer(modifier = Modifier.height(8.dp))

                value = dueDate,
                onValueChange = { dueDate = it },
                label = { Text("Due Date") },
                modifier = Modifier.fillMaxWidth()
            )

            Spacer(modifier = Modifier.height(8.dp))

            Text("Status", fontWeight = FontWeight.Bold)
            Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                listOf("Pending", "Paid", "Overdue").forEach { option ->
                    TextButton(onClick = { status = option }) {
                        Text(if (status == option) "✓ $option" else option)
                    }
                }
            }

            Button(
                enabled = !saving,
                onClick = {
                    val cleanMonth = month.trim()
                    val cleanAmount = amount.trim()
                    val cleanDueDate = dueDate.trim()

                    if (cleanMonth.isEmpty() || cleanAmount.isEmpty() || cleanDueDate.isEmpty()) {
                        onMessage("Please fill Month, Rent Amount and Due Date.")
                        return@Button
                    }

                    if (cleanAmount.toDoubleOrNull() == null || cleanAmount.toDouble() <= 0) {
                        onMessage("Enter a valid rent amount.")
                        return@Button
                    }

                    val duplicate = existingRecords.firstOrNull {
                        it.month.equals(cleanMonth, ignoreCase = true)
                    }

                    if (duplicate != null) {
                        onMessage("A rent record for $cleanMonth already exists for ${resident.fullName}.")
                        return@Button
                    }

                    saving = true
                    val data = hashMapOf(
                        "residentId" to resident.id,
                        "residentName" to resident.fullName,
                        "residentEmail" to resident.email,
                        "room" to resident.room,
                        "bed" to resident.bed,

                        "updatedAt" to FieldValue.serverTimestamp(),

                                resident.id,
                                "Rent Record Updated",
                                "$cleanMonth rent of ₹$cleanAmount is $status. Due date: $cleanDueDate.",
                                "Rent"
                            )
                            saving = false
                            onMessage("Rent record created for ${resident.fullName}.")
                        }
                        .addOnFailureListener { error ->
                            saving = false
                            onMessage("Save failed: ${error.message ?: "Unknown error"}")
                        }
                },
                modifier = Modifier.fillMaxWidth()
            ) {
                Text(if (saving) "Saving..." else "Create Rent Record")
            }

            if (existingRecords.isNotEmpty()) {
                Spacer(modifier = Modifier.height(16.dp))
                Text("Existing Rent Records", fontWeight = FontWeight.Bold)
                Spacer(modifier = Modifier.height(8.dp))

data class AdminSearchResult(
    val type: String,
    val title: String,
    val details: String,
    val status: String = "",
    val category: String = "",
    val date: Date? = null
)

@Composable
fun AdminGlobalSearchScreen(
    onBack: () -> Unit
) {
    val db = FirebaseFirestore.getInstance()

    var query by remember { mutableStateOf("") }
    var selectedType by remember { mutableStateOf("All") }
    var selectedStatus by remember { mutableStateOf("All") }
    var selectedDate by remember { mutableStateOf("All Time") }
    var results by remember { mutableStateOf<List<AdminSearchResult>>(emptyList()) }
    var message by remember { mutableStateOf("Search residents, complaints, services, rent, notices, or rooms.") }
    var isSearching by remember { mutableStateOf(false) }

    val typeFilters = listOf("All", "Resident", "Complaint", "Service", "Rent", "Notice", "Room")
    val statusFilters = listOf("All", "Pending", "In Progress", "Resolved", "Completed", "Paid", "Overdue")
    val dateFilters = listOf("All Time", "Today", "This Week", "This Month")

    fun matches(text: String, search: String): Boolean {
        return search.isBlank() || text.contains(search, ignoreCase = true)
    }

    fun dateMatches(date: Date?, filter: String): Boolean {
        if (filter == "All Time") return true
        if (date == null) return false

        val now = java.util.Calendar.getInstance()
        val target = java.util.Calendar.getInstance().apply { time = date }

        return when (filter) {
            "Today" ->
                now.get(java.util.Calendar.YEAR) == target.get(java.util.Calendar.YEAR) &&
                        now.get(java.util.Calendar.DAY_OF_YEAR) == target.get(java.util.Calendar.DAY_OF_YEAR)

            "This Week" -> {
                val start = java.util.Calendar.getInstance().apply {
                    set(java.util.Calendar.DAY_OF_WEEK, firstDayOfWeek)
                    set(java.util.Calendar.HOUR_OF_DAY, 0)
                    set(java.util.Calendar.MINUTE, 0)
                    set(java.util.Calendar.SECOND, 0)
                    set(java.util.Calendar.MILLISECOND, 0)
                }
                date.time >= start.timeInMillis
            }

            "This Month" ->
                now.get(java.util.Calendar.YEAR) == target.get(java.util.Calendar.YEAR) &&
                        now.get(java.util.Calendar.MONTH) == target.get(java.util.Calendar.MONTH)

            else -> true
        }
    }

    fun documentDate(document: com.google.firebase.firestore.DocumentSnapshot): Date? {
        val timestampFields = listOf("createdAt", "updatedAt", "publishedAt", "date", "dueDate")
        for (field in timestampFields) {
            val timestamp = document.getTimestamp(field)
            if (timestamp != null) return timestamp.toDate()
            val raw = document.getString(field)
            if (!raw.isNullOrBlank()) {
                try {
                    return SimpleDateFormat("yyyy-MM-dd", Locale.getDefault()).parse(raw)
                } catch (_: Exception) {
                    // Try the next supported date field.
                }
            }
        }
        return null
    }

    fun passesFilters(result: AdminSearchResult): Boolean {
        val typeMatches = selectedType == "All" ||
                (selectedType == "Service" && result.type == "Service Request") ||
                result.type == selectedType

        val statusMatches = selectedStatus == "All" ||
                result.status.equals(selectedStatus, ignoreCase = true)

        return typeMatches && statusMatches && dateMatches(result.date, selectedDate)
    }

    fun applyFilters(source: List<AdminSearchResult>) {
        results = source.filter { passesFilters(it) }
        message = if (results.isEmpty()) {
            "No matching results found."
        } else {
            "${results.size} result(s) found."
        }
    }

    fun searchAllCollections() {
        val searchText = query.trim()
        isSearching = true
        message = "Searching..."
        results = emptyList()

        val collections = listOf(
            "residents",
            "complaints",
            "serviceRequests",
            "rentRecords",
            "notices",
            "rooms"
        )

        var completed = 0
        val combinedResults = mutableListOf<AdminSearchResult>()

        collections.forEach { collectionName ->
            db.collection(collectionName).get()
                .addOnSuccessListener { snapshot ->
                    snapshot.documents.forEach { document ->
                        val date = documentDate(document)

                        when (collectionName) {
                            "residents" -> {
                                val name = document.getString("fullName").orEmpty()
                                val email = document.getString("email").orEmpty()
                                val phone = document.getString("phone").orEmpty()
                                val room = document.getString("room").orEmpty()
                                val bed = document.getString("bed").orEmpty()
                                val active = document.getBoolean("active") ?: true
                                val activeText = if (active) "Active" else "Inactive"

                                if (matches(name, searchText) ||
                                    matches(email, searchText) ||
                                    matches(phone, searchText) ||
                                    matches(room, searchText) ||
                                    matches(bed, searchText) ||
                                    matches(activeText, searchText)
                                ) {
                                    combinedResults.add(
                                        AdminSearchResult(
                                            type = "Resident",
                                            title = name.ifEmpty { "Resident" },

@Composable
fun AdminReportsScreen(onBack: () -> Unit) {
    val db = FirebaseFirestore.getInstance()

    var residents by remember { mutableStateOf<List<AnalyticsResident>>(emptyList()) }
    var rents by remember { mutableStateOf<List<AnalyticsRent>>(emptyList()) }
    var rooms by remember { mutableStateOf<List<AnalyticsRoom>>(emptyList()) }

    var selectedYear by remember { mutableStateOf("All Years") }
    var selectedMonth by remember { mutableStateOf("All Months") }
    var selectedRoom by remember { mutableStateOf("All Rooms") }
    var selectedBed by remember { mutableStateOf("All Beds") }

    var yearMenuExpanded by remember { mutableStateOf(false) }
    var monthMenuExpanded by remember { mutableStateOf(false) }
    var roomMenuExpanded by remember { mutableStateOf(false) }
    var bedMenuExpanded by remember { mutableStateOf(false) }

    DisposableEffect(currentPgId()) {
        val residentListener = pgQuery(db, "residents").addSnapshotListener { snapshot, _ ->
            residents = snapshot?.documents.orEmpty().map { d ->
                AnalyticsResident(
                    id = d.id,
                    name = d.getString("fullName").orEmpty(),
                    room = d.getString("room").orEmpty(),
                    bed = d.getString("bed").orEmpty(),
                    status = d.getString("status") ?: "Pending"
                )
            }
        }
        val rentListener = pgQuery(db, "rentRecords").addSnapshotListener { snapshot, _ ->
            rents = snapshot?.documents.orEmpty().map { d ->
                AnalyticsRent(
                    month = d.getString("month").orEmpty(),
                    amount = firestoreNumber(d.get("amount")),
                    status = d.getString("status") ?: "Pending",
                    room = d.getString("room").orEmpty(),
                    bed = d.getString("bed").orEmpty()
                )
            }
        }
        val roomListener = pgQuery(db, "rooms").addSnapshotListener { snapshot, _ ->
            rooms = snapshot?.documents.orEmpty().map { d ->
                AnalyticsRoom(
                    roomNumber = d.getString("roomNumber") ?: d.id,
                    totalBeds = (d.getLong("totalBeds") ?: 0L).toInt(),
                    occupiedBeds = (d.getLong("occupiedBeds") ?: 0L).toInt()
                )
            }.sortedBy { it.roomNumber }
        }
        onDispose {
            residentListener.remove()
            rentListener.remove()
            roomListener.remove()
        }
    }

    val years = rents.mapNotNull { analyticsYear(it.month) }
        .distinct()
        .sortedDescending()
    val monthsForSelectedYear = rents
        .filter {
            selectedYear == "All Years" ||
                    analyticsYear(it.month)?.toString() == selectedYear
        }
        .mapNotNull { analyticsMonthName(it.month) }
        .distinct()
        .sortedBy { analyticsMonthNumber(it) }

    val months = listOf("All Months") + monthsForSelectedYear

    // Start with the newest available year/month when records first arrive.
    var autoSelectedLatestPeriod by remember { mutableStateOf(false) }
    LaunchedEffect(rents) {
        if (!autoSelectedLatestPeriod && rents.isNotEmpty()) {
            val latest = rents.maxByOrNull { analyticsPeriodKey(it.month) }
            if (latest != null) {
                selectedYear = analyticsYear(latest.month)?.toString() ?: "All Years"
                selectedMonth = analyticsMonthName(latest.month) ?: "All Months"
            }
            autoSelectedLatestPeriod = true
        }
    }

    if (selectedYear != "All Years" && selectedYear !in years.map { it.toString() }) {
        selectedYear = "All Years"
    }
    if (selectedMonth != "All Months" && selectedMonth !in months) {
        selectedMonth = "All Months"
    }

    val activeResidents = residents.filter { it.status.trim().equals("Active", true) }

    val periodFilteredRents = rents.filter { rent ->
        val yearMatches = selectedYear == "All Years" || analyticsYear(rent.month)?.toString() == selectedYear
        val monthMatches = selectedMonth == "All Months" || analyticsMonthName(rent.month).equals(selectedMonth, true)
        yearMatches && monthMatches
    }
    val roomFilteredRents = periodFilteredRents.filter {
        selectedRoom == "All Rooms" || it.room.trim().equals(selectedRoom.trim(), true)
    }
    val filteredRents = roomFilteredRents.filter {
        selectedBed == "All Beds" || it.bed.trim().equals(selectedBed.trim(), true)
    }

    val roomResidents = activeResidents.filter {
        selectedRoom == "All Rooms" || it.room.trim().equals(selectedRoom.trim(), true)
    }
    val filteredResidents = roomResidents.filter {
        selectedBed == "All Beds" || it.bed.trim().equals(selectedBed.trim(), true)
    }

    val expected = filteredRents.sumOf { it.amount }
    val collected = filteredRents.filter { it.status.trim().equals("Paid", true) }.sumOf { it.amount }
    val pending = filteredRents.filter { it.status.trim().equals("Pending", true) }.sumOf { it.amount }
    val overdue = filteredRents.filter { it.status.trim().equals("Overdue", true) }.sumOf { it.amount }
    val paidCount = filteredRents.count { it.status.trim().equals("Paid", true) }
    val pendingCount = filteredRents.count { it.status.trim().equals("Pending", true) }
    val overdueCount = filteredRents.count { it.status.trim().equals("Overdue", true) }
    val collectionRate = if (expected > 0) collected / expected * 100.0 else 0.0

    val availableBeds = rooms.firstOrNull {
        selectedRoom != "All Rooms" && it.roomNumber.equals(selectedRoom, true)
    }?.let { (it.totalBeds - it.occupiedBeds).coerceAtLeast(0) }
        ?: rooms.sumOf { (it.totalBeds - it.occupiedBeds).coerceAtLeast(0) }

    val bedsForSelectedRoom = if (selectedRoom == "All Rooms") {
        activeResidents.map { it.bed }
            .filter { it.isNotBlank() && it != "Not Assigned" }
            .distinct()
            .sorted()
    } else {
        activeResidents.filter { it.room.equals(selectedRoom, true) }
            .map { it.bed }
            .filter { it.isNotBlank() && it != "Not Assigned" }
            .distinct()
            .sorted()
    }

    LaunchedEffect(selectedRoom, bedsForSelectedRoom) {
        if (selectedBed != "All Beds" && selectedBed !in bedsForSelectedRoom) {
            selectedBed = "All Beds"
        }
    }

    // Keep this chart independent from the fee-status filter so it always shows
    // the complete month-by-month collection history for the current PG.
    val monthTotals = rents.groupBy { it.month }
        .map { (month, records) ->
            AnalyticsMonth(
                month = month,
                expected = records.sumOf { it.amount },
                collected = records.filter { it.status.trim().equals("Paid", true) }.sumOf { it.amount }
            )
        }
        .sortedBy { analyticsPeriodKey(it.month) }

    Column(modifier = Modifier.fillMaxSize().padding(20.dp)) {
        TextButton(onClick = onBack) { Text("← Back") }
        Text("Reports & Analytics", fontSize = 28.sp, fontWeight = FontWeight.Bold)
        Text("PG: ${currentPgId()}", fontSize = 14.sp)
        Spacer(modifier = Modifier.height(12.dp))

        Text("Fee Filters", fontSize = 19.sp, fontWeight = FontWeight.Bold)
        Spacer(modifier = Modifier.height(8.dp))

        AnalyticsDropdown(
            "Year",
            selectedYear,
            listOf("All Years") + years.map { it.toString() },
            yearMenuExpanded,
            { yearMenuExpanded = it }
        ) {
            selectedYear = it
            selectedMonth = "All Months"
            selectedBed = "All Beds"
            yearMenuExpanded = false
        }

        Spacer(modifier = Modifier.height(8.dp))
        AnalyticsDropdown(
            "Month",
            selectedMonth,
            months,
            monthMenuExpanded,
            { monthMenuExpanded = it }
        ) {
            selectedMonth = it
            monthMenuExpanded = false
        }

        Spacer(modifier = Modifier.height(8.dp))
        AnalyticsDropdown(
            "Room",
            selectedRoom,
            listOf("All Rooms") + rooms.map { it.roomNumber },
            roomMenuExpanded,
            { roomMenuExpanded = it }
        ) {
            selectedRoom = it
            selectedBed = "All Beds"
            roomMenuExpanded = false
        }

        Spacer(modifier = Modifier.height(8.dp))
        AnalyticsDropdown(
            "Bed",
            selectedBed,
            listOf("All Beds") + bedsForSelectedRoom,
            bedMenuExpanded,
            { bedMenuExpanded = it }
        ) {
            selectedBed = it
            bedMenuExpanded = false
        }

        Spacer(modifier = Modifier.height(10.dp))
        Text(
            "Showing: $selectedYear • $selectedMonth • $selectedRoom • $selectedBed",
            fontSize = 13.sp,
            color = Color.Gray
        )

        Spacer(modifier = Modifier.height(16.dp))
        LazyColumn(verticalArrangement = Arrangement.spacedBy(12.dp), modifier = Modifier.fillMaxSize()) {
            item {
                AnalyticsCard("Residents", listOf(
                    "Active Residents: ${filteredResidents.size}",
                    "Available Beds: $availableBeds"
                ))
            }
            item {
                AnalyticsCard("Fee Summary", listOf(
                    "Expected Fee: ₹${String.format(Locale.getDefault(), "%,.0f", expected)}",
                    "Collected: ₹${String.format(Locale.getDefault(), "%,.0f", collected)}",
                    "Pending: ₹${String.format(Locale.getDefault(), "%,.0f", pending)}",
                    "Overdue: ₹${String.format(Locale.getDefault(), "%,.0f", overdue)}",
                    "Collection Rate: ${String.format(Locale.getDefault(), "%.1f", collectionRate)}%"
                ))
            }
            item {
                // Key forces the Canvas to be recreated whenever the filtered
                // Paid/Pending/Overdue counts change, preventing stale slices.
                key(selectedYear, selectedMonth, selectedRoom, selectedBed, paidCount, pendingCount, overdueCount) {
                    FeeStatusPieChart(paidCount, pendingCount, overdueCount)
                }
            }
            item {
                AnalyticsCard("Student Fee Status", listOf(
                    "Paid Students: $paidCount",
                    "Pending Students: $pendingCount",
                    "Overdue Students: $overdueCount",
                    "Total Fee Records: ${filteredRents.size}"
                ))
            }
            item {
                MonthlyCollectionChart(monthTotals)
            }
            item {
                AnalyticsCard("Room & Bed Overview", rooms.map {
                    val available = (it.totalBeds - it.occupiedBeds).coerceAtLeast(0)
                    "Room ${it.roomNumber}: ${it.occupiedBeds}/${it.totalBeds} occupied • $available available"
                }.ifEmpty { listOf("No rooms created for this PG yet.") })
            }
            item {
                Text("Analytics update automatically when Firestore data changes.", fontSize = 14.sp)
            }
            item { Spacer(modifier = Modifier.height(20.dp)) }
        }
    }
}

data class AnalyticsResident(val id: String, val name: String, val room: String, val bed: String, val status: String)
data class AnalyticsRent(val month: String, val amount: Double, val status: String, val room: String, val bed: String)
data class AnalyticsRoom(val roomNumber: String, val totalBeds: Int, val occupiedBeds: Int)
data class AnalyticsMonth(val month: String, val expected: Double, val collected: Double)

private fun analyticsYear(value: String): Int? {
    val match = Regex("\\b(19|20)\\d{2}\\b").find(value.trim()) ?: return null
    return match.value.toIntOrNull()
}

private fun analyticsMonthName(value: String): String? {
    val cleaned = value.trim()
    if (cleaned.isBlank()) return null
    val formats = listOf("MMMM yyyy", "MMM yyyy", "MMMM, yyyy", "MMM, yyyy")
    for (format in formats) {
        try {
            val parsed = SimpleDateFormat(format, Locale.ENGLISH).apply { isLenient = false }.parse(cleaned)
            if (parsed != null) return SimpleDateFormat("MMMM", Locale.ENGLISH).format(parsed)
        } catch (_: Exception) {
        }
    }
    return cleaned.substringBeforeLast(" ").takeIf { it.isNotBlank() }
}

private fun analyticsMonthNumber(value: String): Int {
    return try {
        SimpleDateFormat("MMMM", Locale.ENGLISH).parse(value)?.let {
            SimpleDateFormat("M", Locale.ENGLISH).format(it).toInt()
        } ?: 99
    } catch (_: Exception) {
        99
    }
}

private fun analyticsPeriodKey(value: String): Long {
    val year = analyticsYear(value) ?: 0
    val month = analyticsMonthName(value)?.let { analyticsMonthNumber(it) } ?: 0
    return year.toLong() * 100L + month.toLong()
}

@Composable
private fun AnalyticsDropdown(label: String, value: String, options: List<String>, expanded: Boolean, setExpanded: (Boolean) -> Unit, onSelect: (String) -> Unit) {

        )
        DropdownMenu(
            expanded = expanded,
            onDismissRequest = { setExpanded(false) },
            modifier = Modifier.fillMaxWidth()
        ) {
            options.distinct().forEach { option ->
                DropdownMenuItem(text = { Text(option) }, onClick = { onSelect(option) })
            }
        }
    }
}

@Composable
private fun FeeStatusPieChart(paid: Int, pending: Int, overdue: Int) {
    val safePaid = paid.coerceAtLeast(0)
    val safePending = pending.coerceAtLeast(0)
    val safeOverdue = overdue.coerceAtLeast(0)
    val total = safePaid + safePending + safeOverdue

    Card(modifier = Modifier.fillMaxWidth(), elevation = CardDefaults.cardElevation(defaultElevation = 3.dp)) {
        Column(modifier = Modifier.padding(16.dp), horizontalAlignment = Alignment.CenterHorizontally) {
            Text("Fee Status", fontSize = 20.sp, fontWeight = FontWeight.Bold)
            Spacer(modifier = Modifier.height(12.dp))

            if (total == 0) {
                Canvas(modifier = Modifier.height(210.dp).fillMaxWidth()) {
                    val diameter = minOf(size.width, size.height) * 0.72f
                    val left = (size.width - diameter) / 2f
                    val top = (size.height - diameter) / 2f
                    drawArc(
                        color = Color.LightGray,
                        startAngle = 0f,
                        sweepAngle = 360f,
                        useCenter = true,
                        topLeft = androidx.compose.ui.geometry.Offset(left, top),
                        size = androidx.compose.ui.geometry.Size(diameter, diameter)
                    )
                }
                Text("No data", fontWeight = FontWeight.SemiBold)
                Spacer(modifier = Modifier.height(8.dp))
                Text("No fee records for this filter.", fontSize = 14.sp)
            } else {
                val slices = listOf(
                    PieSlice("Paid", safePaid, Color(0xFF2E7D32)),
                    PieSlice("Pending", safePending, Color(0xFFF9A825)),
                    PieSlice("Overdue", safeOverdue, Color(0xFFC62828))
                )

                Canvas(modifier = Modifier.height(210.dp).fillMaxWidth()) {
                    val diameter = minOf(size.width, size.height) * 0.72f
                    val left = (size.width - diameter) / 2f
                    val top = (size.height - diameter) / 2f
                    val arcSize = androidx.compose.ui.geometry.Size(diameter, diameter)
                    val arcOffset = androidx.compose.ui.geometry.Offset(left, top)
                    var startAngle = -90f

                    slices.forEach { slice ->
                        if (slice.value > 0) {
                            val sweepAngle = slice.value.toFloat() / total.toFloat() * 360f
                            drawArc(
                                color = slice.color,
                                startAngle = startAngle,
                                sweepAngle = sweepAngle,
                                useCenter = true,
                                topLeft = arcOffset,
                                size = arcSize
                            )
                            startAngle += sweepAngle
                        }
                    }
                }

                Spacer(modifier = Modifier.height(8.dp))
                slices.forEach { slice ->
                    val percentage = slice.value.toDouble() / total.toDouble() * 100.0
                    Row(
                        modifier = Modifier.fillMaxWidth().padding(vertical = 3.dp),
                        verticalAlignment = Alignment.CenterVertically
                    ) {
                        Box(
                            modifier = Modifier
                                .size(12.dp)
                                .background(slice.color, RoundedCornerShape(3.dp))
                        )
                        Spacer(modifier = Modifier.width(8.dp))
                        Text(
                            "${slice.label}: ${slice.value} (${String.format(Locale.getDefault(), "%.1f", percentage)}%)",
                            fontSize = 14.sp
                        )
                    }
                }
            }
        }
    }
}

@Composable
private fun MonthlyCollectionChart(data: List<AnalyticsMonth>) {
    Card(modifier = Modifier.fillMaxWidth(), elevation = CardDefaults.cardElevation(defaultElevation = 3.dp)) {
        Column(modifier = Modifier.padding(16.dp)) {
            Text("Month-by-Month Collection", fontSize = 20.sp, fontWeight = FontWeight.Bold)
            Spacer(modifier = Modifier.height(12.dp))
            if (data.isEmpty()) {
                Text("No monthly rent records available yet.")
            } else {
                val maxValue = data.maxOf { maxOf(it.expected, it.collected) }.coerceAtLeast(1.0)
                data.takeLast(12).forEach { item ->
                    Text(item.month, fontWeight = FontWeight.SemiBold)
                    Spacer(modifier = Modifier.height(3.dp))
                    Text("Collected ₹${String.format(Locale.getDefault(), "%,.0f", item.collected)} / Expected ₹${String.format(Locale.getDefault(), "%,.0f", item.expected)}", fontSize = 13.sp)
                    Spacer(modifier = Modifier.height(10.dp))
                }
            }
        }
    }
}
